## 207. 课程表

```python
class Solution:
    def canFinish(self, numCourses, prerequisites):
        # 1. 构建邻接表 & 入度表
        adjacency = defaultdict(list)
        in_degree = [0] * numCourses
       
        for dest, src in prerequisites:
            adjacency[src].append(dest)
            in_degree[dest] += 1
       
        # 2. 初始化队列（入度为0的课程）
        queue = deque()
        learned_count = 0
        for i in range(numCourses):
            if in_degree[i] == 0:
                queue.append(i)
               
         # 3. Kahn 拓扑排序
        while queue:
            course = queue.popleft()
            learned_count += 1
           
            for next_course in adjacency[course]:
                in_degree[next_course] -= 1
                if in_degree[next_course] == 0:
                    queue.append(next_course)
                   
        # 4. 如果学完的课程数等于总课程数 → 没有环
        return learned_count == numCourses
```

## **208. 实现 Trie (前缀树)**

```python
class Trie:
    def __init__(self):
        self.children = [None] * 26
        self.isEnd = False
    
    def searchPrefix(self, prefix: str) -> "Trie":
        current = self
        for ch in prefix:
            index = ord(ch) - ord("a")
            if not current.children[index]:
                return None
            current = current.children[index]
        return current

    def insert(self, word: str) -> None:
        current = self  # 当前位置 根节点
        for char in word:
            index = ord(char) - ord("a")
            if not current.children[index]:
                current.children[index] = Trie()
            current = current.children[index]  # 移动到下一个位置
        current.isEnd = True

    def search(self, word: str) -> bool:
        node = self.searchPrefix(word)
        return node is not None and node.isEnd

    def startsWith(self, prefix: str) -> bool:
        return self.searchPrefix(prefix) is not None
```

## 139. 单词拆分

**DFS 暴力**

```python
# 从左往右“拿剪刀”切，如果前面一段在字典里，就递归处理剩下的
# 如果最后能切到结尾，说明成功
def wordBreak_dfs(s, wordDict):
    wordSet = set(wordDict)  # 为了加速查找
    n = len(s)

    def dfs(start):
        if start == n:  # 如果切到结尾
            return True
        for end in range(start + 1, n + 1):  # 从 start 往后切
            if s[start:end] in wordSet:  # 如果这段是单词
                if dfs(end):  # 继续切后面
                    return True
        return False

    return dfs(0)
```

**DFS + 记忆化 (memo)**

```python
# 给每个 start 位置加一个备忘录
# 如果从这个位置开始算过，就直接用结果，不要重复算
def wordBreak_memo(s, wordDict):
    wordSet = set(wordDict)
    n = len(s)
    memo = {}

    def dfs(start):
        if start == n:
            return True
        if start in memo:   # 加记忆化
            return memo[start]

        for end in range(start + 1, n + 1):
            if s[start:end] in wordSet:
                if dfs(end):
                    # 这行代码并不会起到帮助(暂时这么理解)
                    # 一条最终失败的路径，其本身不可能导致 memo[start] = True 被设置
                    # memo[start] = True 只可能在 dfs(start) 这个子问题被成功解决时才会发生
                    # 如果一条路径最终失败了，说明这条路径上的某个环节返回了 False
                    # 这个 False 会沿着调用栈向上传播。所以，我的那个“失败路径中计算出 True”的假设是错误的
                    # memo[start] = True
                    return True

        memo[start] = False
        return False

    return dfs(0)
```

**动态规划**

```python
def wordBreak(s, wordDict):
    wordSet = set(wordDict)
    dp = [False] * (len(s) + 1)
    dp[0] = True

    for i in range(1, len(s) + 1): # i 表示右边界（切片的结束位置）
        for j in range(i):  # j 表示左边界（切片的起始位置）
            if dp[j] and s[j:i] in wordSet:
                dp[i] = True
                break
    return dp[len(s)]
```

如果 `for i in range(1, len(s))` 没有加1
最大的 `i = len(s) - 1`
→ 永远不会计算 `dp[len(s)]`
→ 最后一段（包含最后一个字符）不会被考虑到

因为我们用了两个  `range`

`for i in range(1, len(s) + 1)`:
        `for j in range(i)`:

如果不 `+1`, 因为 左闭右开, j 就不能到达最后一个
